CREATE TABLE ASSETS_per_USER_DEPARTMENT
(
    asset_per_user_department_id INT IDENTITY(1,1) PRIMARY KEY,

    Tagging_id INT NOT NULL,
    User_department_id INT NOT NULL,
    Assets_id INT NOT NULL,
    Assets_category_id INT NOT NULL,

    Assets_Quantity INT NOT NULL DEFAULT 1,

    IsActive BIT NOT NULL DEFAULT 1,
    Created_at DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    Updated_at DATETIME2 NULL,
    last_checked_at DATETIME2 NULL,

    Asset_Model VARCHAR(100) NULL,
    Asset_Serial_Number VARCHAR(100) NULL,

    Purchase_Date DATE NULL,
    Purchase_Cost DECIMAL(18,2) NULL,
    Remarks VARCHAR(MAX) NULL,

    CONSTRAINT FK_AssetsPerUserDepartment_Tagging 
    FOREIGN KEY (Tagging_id)
    REFERENCES TAGGING(Tagging_id)
    ON UPDATE CASCADE
    ON DELETE NO ACTION,
    
    CONSTRAINT FK_AssetsPerUserDepartment_UserDepartment
    FOREIGN KEY (User_department_id)
    REFERENCES USERS(User_department_id)
    ON UPDATE CASCADE
    ON DELETE NO ACTION,

    CONSTRAINT FK_AssetsPerUserDepartment_Assets
    FOREIGN KEY (Assets_id)
    REFERENCES ASSETS(Assets_id)
    ON UPDATE CASCADE
    ON DELETE NO ACTION,

    CONSTRAINT FK_AssetsPerUserDepartment_AssetsCategory
    FOREIGN KEY (Assets_category_id)
    REFERENCES ASSETS_CATEGORY(Assets_category_id)
    ON UPDATE CASCADE
    ON DELETE NO ACTION
);

-- Unique serial number kapag may actual value.

CREATE UNIQUE INDEX UX_AssetsPerUserDepartment_SerialNumber
ON ASSETS_per_USER_DEPARTMENT (Asset_Serial_Number)
WHERE Asset_Serial_Number IS NOT NULL;


































DISPLAYING QUERY



SELECT
CONCAT(u.User_fname, ' ', u.User_lname) as User_Fullname,
d.Department_name,
div.Division_name,
apud.Asset_Model,
apud.Asset_Serial_Number,
apud.Purchase_Date,
apud.Purchase_Cost,
ac.Assets_category_name,
ac.Assets_category_description,
a.Asset_name,
a.Asset_description,
t.performance_status,
t.Tagging_description
FROM USERS u
INNER JOIN Departments d
ON u.Department_id = d.Department_id
INNER JOIN Departments_per_Division dpb
ON dpb.Department_id = d.Department_id
INNER JOIN Divisions div
ON div.Division_id = dpb.Division_id
LEFT JOIN ASSETS_per_USER_DEPARTMENT apud
ON apud.User_department_id = u.User_department_id
INNER JOIN ASSETS_CATEGORY ac
ON apud.Assets_category_id = ac.Assets_category_id
INNER JOIN ASSETS a
ON apud.Assets_id = a.Assets_id
INNER JOIN TAGGING t
ON apud.Tagging_id = t.Tagging_id



WHERE u.IsActive = 1 ORDER BY User_Fullname ASC;



SELECT
    -- Asset Assignment ID
    apud.asset_per_user_department_id,

    -- Asset Information
    a.Assets_id,
    a.Asset_name,
    a.Asset_description,

    -- Category
    ac.Assets_category_id,
    ac.Assets_category_name,
    ac.Assets_category_description,

    -- Owner / Assigned User
    u.User_department_id,
    u.User_fname,
    u.User_middlename,
    u.User_lname,
    CONCAT(
        u.User_fname,
        ' ',
        ISNULL(u.User_middlename + ' ', ''),
        u.User_lname
    ) AS User_Fullname,

    -- Department
    d.Department_id,
    d.Department_code,
    d.Department_name,

    -- Division
    div.Division_id,
    div.Division_code,
    div.Division_name,

    -- Asset Condition / Tagging
    t.Tagging_id,
    t.performance_status,
    t.Tagging_description,
    t.efficiency_percentage,
    t.is_operational,

    -- Physical Asset Details
    apud.Asset_Model,
    apud.Asset_Serial_Number,
    apud.Assets_Quantity,

    -- Purchase Details
    apud.Purchase_Date,
    apud.Purchase_Cost,

    -- Other Information
    apud.Remarks,
    apud.last_checked_at,
    apud.IsActive,
    apud.Created_at,
    apud.Updated_at

FROM ASSETS_per_USER_DEPARTMENT apud

INNER JOIN ASSETS a
    ON apud.Assets_id = a.Assets_id

INNER JOIN ASSETS_CATEGORY ac
    ON apud.Assets_category_id = ac.Assets_category_id

INNER JOIN USERS u
    ON apud.User_department_id = u.User_department_id

INNER JOIN Departments d
    ON u.Department_id = d.Department_id

INNER JOIN Departments_per_Division dpb
    ON dpb.Department_id = d.Department_id

INNER JOIN Divisions div
    ON div.Division_id = dpb.Division_id

INNER JOIN TAGGING t
    ON apud.Tagging_id = t.Tagging_id

WHERE apud.asset_per_user_department_id = 1;
